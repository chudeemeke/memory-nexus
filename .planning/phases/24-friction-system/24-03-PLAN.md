case "dashboard": {
  const stats = await service.getStats();
  const trends = await service.getWeeklyTrends(12);
  const openItems = await service.list();  // open items only

  if (options.html) {
    // Generate HTML and write to ~/.memory/dashboard.html
    const html = generateFrictionHtml(stats, trends, openItems);
    const memoryDir = getMemoryDir();
    mkdirSync(memoryDir, { recursive: true });  // ensure ~/.memory/ exists (Pitfall 6)
    const dashboardPath = join(memoryDir, "dashboard.html");
    writeFileSync(dashboardPath, html, "utf-8");

    if (!options.json) {
      console.log(`Dashboard written to ${dashboardPath}`);
      // Open in default browser
      openInBrowser(dashboardPath);
    }
  } else if (options.json) {
    console.log(JSON.stringify({ stats, trends }, null, 2));
  } else {
    // Rich terminal output
    const output = formatFrictionDashboard(stats, trends, openItems, shouldUseColor());
    console.log(output);
  }
  break;
}
```

import { exec } from "node:child_process";
import { platform } from "node:os";

function openInBrowser(filePath: string): void {
  const cmd = platform() === "win32" ? "start" :
              platform() === "darwin" ? "open" : "xdg-open";
  exec(`${cmd} "${filePath}"`);
}
```

## Friction Logging

When you encounter friction with the memory tool (commands that fail, output that is unhelpful, missing features, workarounds needed), log it:

```bash
memory friction log "<description>" --severity <low|medium|high|critical> --category <search|sync|cli|context|integration|ux>
```

Severity guide:
- **critical**: Tool is broken, blocks work
- **high**: Significant friction, workaround needed
- **medium**: Annoying but manageable
- **low**: Minor inconvenience, nice-to-fix

Review friction periodically:
```bash
memory friction list                    # Open items
memory friction dashboard              # Visual overview
memory friction dashboard --html       # Detailed HTML report
memory friction resolve <id> --resolution "how it was fixed"
```
```

    constructor(
        private readonly smartContext: SmartContextService,
        private readonly contextWriter: IAmbientContextWriter,

    ) {}

    async generateAmbientContext(options: AmbientContextOptions): Promise<AmbientContextResult> { ... }

**Implementation logic:**

1. Call `this.smartContext.getContext({ projectFilter: options.projectName, budget: options.budget, crossProject: true })`.
2. If result is null, return `{ success: false, reason: "project-not-found" }`.
3. If result has zero sections, return `{ success: false, reason: "no-context" }`.
4. Format the result for context.md using `this.formatter.formatSmartContext(result)`.
5. Build the MEMORY.md summary block using `this.buildSummaryBlock(result)`.
6. Call `this.contextWriter.writeContextFile(options.autoMemoryDir, contextContent)`.
7. Call `this.contextWriter.updateMemoryBlock(options.autoMemoryDir, summaryBlock)`.
8. Return `{ success: true, contextTokens: result.totalTokensEstimate }`.

**buildSummaryBlock(result) private method:**

Produce a concise summary matching the CONTEXT.md-specified format:

```markdown
## Cross-Project Context
Run `memory context <project>` for full briefing. See [context.md](context.md) for latest snapshot.
- {count} active decisions, {count} learnings
- Open friction: {count summary}
- Last synced: {YYYY-MM-DD}
```

Extract counts from the SmartContextResult sections:
- Count "decisions" section lines (approximate active decisions count)
- Count "learnings" section lines
- Count "friction" section lines
- Use the current date for "Last synced"

Keep the block under 10 lines per the CONTEXT.md spec.

Add a new async function `runAmbientContextGeneration`:

```typescript
/**
 * Generate ambient context files for the current project.
 *
 * Runs after memory file sync. Writes context.md and updates MEMORY.md
 * in the current project's Claude Code auto memory directory.
 *
 * Failure is non-fatal: logs error and continues sync.
 */
async function runAmbientContextGeneration(
    db: ReturnType<typeof initializeDatabase>["db"],
    options: SyncCommandOptions,
): Promise<void> {
    try {
        // Load config (lazy -- only when ambient context is needed)
        const { loadConfig } = await import("../../../infrastructure/hooks/config-manager.js");
        const config = loadConfig();

        if (!config.ambientContext.enabled) {
            return; // Disabled by config
        }

        // Resolve auto memory directory from cwd
        const cwd = process.cwd();
        const { ProjectPath } = await import("../../../domain/value-objects/project-path.js");
        const encoded = ProjectPath.fromDecoded(cwd).encoded;
        const { homedir } = await import("node:os");
        const { join } = await import("node:path");
        const autoMemoryDir = join(homedir(), ".claude", "projects", encoded, "memory");

        // Resolve project name from path
        const projectName = ProjectPath.fromDecoded(cwd).projectName;

        // Create dependencies
        const { SqliteProjectResolver } = await import(
            "../../../infrastructure/database/services/context-service.js"
        );
        const { SqliteMemoryFileRepository } = await import(
            "../../../infrastructure/database/repositories/memory-file-repository.js"
        );
        const { SqliteFrictionRepository } = await import(
            "../../../infrastructure/database/repositories/friction-repository.js"
        );
        const { AutoMemoryWriter } = await import(
            "../../../infrastructure/hooks/auto-memory-writer.js"
        );
        const { SmartContextService } = await import(
            "../../../application/services/smart-context-service.js"
        );
        const { AmbientContextService } = await import(
            "../../../application/services/ambient-context-service.js"
        );
        const { createContextFormatter } = await import(
            "../formatters/context-formatter.js"
        );

        const projectResolver = new SqliteProjectResolver(db);
        const memoryFileRepo = new SqliteMemoryFileRepository(db);
        const frictionRepo = new SqliteFrictionRepository(db);
        const formatter = createContextFormatter("ai", false);

        const smartContext = new SmartContextService({
            projectResolver,
            memoryFileRepo,
            frictionRepo,
        });

        const ambientService = new AmbientContextService(
            smartContext,
            new AutoMemoryWriter(),
            formatter as { formatSmartContext(result: any): string },
        );

        const result = await ambientService.generateAmbientContext({
            projectName,
            autoMemoryDir,
            budget: config.ambientContext.budget,
        });

        if (result.success && !options.quiet) {
            console.log(`  Ambient context: updated (~${result.contextTokens} tokens)`);
        }
    } catch (error) {
        // Non-fatal: ambient context generation should never fail the sync
        if (!options.quiet) {
            console.error(
                `  Ambient context: error (${error instanceof Error ? error.message : String(error)})`
            );
        }
    }
}
```

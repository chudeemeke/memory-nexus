```typescript
// src/presentation/cli/formatters/envelope.ts (NEW)

export const ENVELOPE_SCHEMA_VERSION = "1" as const;

export type QueryCommandName =
  | "search" | "context" | "show" | "list" | "related" | "stats";
export type QueryResultKind =
  | "message" | "session" | "context" | "related" | "stats";

export interface QueryResultEnvelope<T = unknown> {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  kind: QueryResultKind;
  scope?: string;
  meta?: Record<string, unknown>;
  data: T;
}

export interface QueryErrorEnvelope {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  error: {
    code: string;
    message: string;
    context?: Record<string, unknown>;
  };
}

export function buildEnvelope<T>(args: {
  command: QueryCommandName;
  kind: QueryResultKind;
  data: T;
  scope?: string;
  meta?: Record<string, unknown>;
}): QueryResultEnvelope<T>;

export function buildErrorEnvelope(args: {
  command: QueryCommandName;
  code: string;
  message: string;
  context?: Record<string, unknown>;
}): QueryErrorEnvelope;
```

<plan_structure_rationale>
Plan 32-01 is split from 32-02 and 32-03 because:
1. **Single concern** — this plan establishes contracts; downstream plans consume them.
2. **No per-command file churn** — keeps `files_modified` non-overlapping with 32-02 and 32-03, so 32-02/32-03 can read the envelope module without race conditions on the same file.
3. **Wave 1 only** — no dependencies. Plans 32-02 and 32-03 enter Wave 2 because they both consume `envelope.ts` produced here.

The alternative (single fat PLAN) would create a 7-task plan covering envelope + groups + 6 command refactors + format normalization — well past the 2-3 task budget and >50% context.
</plan_structure_rationale>

<phase_32_5_convergence>
This plan's envelope deliberately carries `kind` and reserves `scope` so Phase 32.5's unified query primitive (`memory query --scope <project|global> --kind <message|session|...>`) can route results through the SAME envelope shape with no breaking change. Per audit §21 acceptance: "Read surfaces (search / context / related / list / show) are accessible via ONE query primitive with shape flags." Phase 32 lays the field; Phase 32.5 sets the field from a CLI flag instead of the command name. No schema bump expected.
</phase_32_5_convergence>

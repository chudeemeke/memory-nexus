```typescript
// src/presentation/cli/formatters/envelope.ts (NEW)

export const ENVELOPE_SCHEMA_VERSION = "1" as const;

// Per Codex HIGH-1: define `as const` tuples (runtime values), derive types from them.
// Bare `type QueryCommandName = "search" | ...` cannot be "exported as a value" — it must
// be derived from a runtime tuple for tests to assert against `QUERY_COMMAND_NAMES.includes(x)`.
export const QUERY_COMMAND_NAMES = [
  "search",
  "context",
  "show",
  "list",
  "related",
  "stats",
] as const;
export type QueryCommandName = typeof QUERY_COMMAND_NAMES[number];

export const QUERY_RESULT_KINDS = [
  "message",
  "session",
  "context",
  "related",
  "stats",
  "file",
] as const;
export type QueryResultKind = typeof QUERY_RESULT_KINDS[number];

// Per Codex MEDIUM-4: discriminated scope shape (NOT freeform string) so Phase 32.5's
// `--scope <project|global>` maps 1:1 to the envelope field without ambiguity.
export type EnvelopeScope =
  | { type: "global" }
  | { type: "project"; project: string };

export interface QueryResultEnvelope<T = unknown> {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  kind: QueryResultKind;
  scope?: EnvelopeScope;
  meta?: Record<string, unknown>;
  data: T;
}

export interface QueryErrorEnvelope {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  error: {
    code: string;
    message: string;
    context?: Record<string, unknown>;
  };
}

export function buildEnvelope<T>(args: {
  command: QueryCommandName;
  kind: QueryResultKind;
  data: T;
  scope?: EnvelopeScope;
  meta?: Record<string, unknown>;
}): QueryResultEnvelope<T>;

export function buildErrorEnvelope(args: {
  command: QueryCommandName;
  code: string;
  message: string;
  context?: Record<string, unknown>;
}): QueryErrorEnvelope;

// Per Codex HIGH-2: shared emission helpers. EVERY exit path in Plan 02 routes through
// these — success, empty-result, validation-error, not-found, catch. No bespoke
// JSON.stringify calls in command files. Centralizes stdout/exitCode contract.
export function emitJsonEnvelope<T>(args: {
  command: QueryCommandName;
  kind: QueryResultKind;
  data: T;
  scope?: EnvelopeScope;
  meta?: Record<string, unknown>;
}): void;
// Writes JSON.stringify(buildEnvelope(args), null, 2) to stdout.
// Does NOT set exitCode (caller still returns { exitCode: 0 }).

export function emitJsonErrorEnvelope(args: {
  command: QueryCommandName;
  code: string;
  message: string;
  context?: Record<string, unknown>;
}): void;
// Writes JSON.stringify(buildErrorEnvelope(args), null, 2) to stdout.
// Does NOT set exitCode (caller still returns { exitCode: 1 }).
```

1. **Single concern** — this plan establishes contracts; downstream plans consume them.

</plan_structure_rationale>

<phase_32_5_convergence>
This plan's envelope deliberately carries `kind` and reserves `scope` (now as a discriminated shape per MEDIUM-4) so Phase 32.5's unified query primitive (`memory query --scope <project|global> --kind <message|session|...>`) can route results through the SAME envelope shape with no breaking change. Per audit §21 acceptance: "Read surfaces (search / context / related / list / show) are accessible via ONE query primitive with shape flags." Phase 32 lays the field; Phase 32.5 sets the field from a CLI flag. The discriminated `scope` shape (object, not string) maps 1:1 to `--scope global|project [--project <name>]`. No schema bump expected.
</phase_32_5_convergence>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| CLI arguments → program parser | User input parsed by Commander; `.choices()` validates values |
| Program help output → terminal | No user data emitted; help text is static |
| Envelope helpers → caller (Plans 32-02, 32-03) | Caller controls `data` field; envelope is structural wrapper |
| Emission helpers → stdout | `console.log` is the canonical write surface; `JSON.stringify` escapes user data |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-32-01 | Tampering | envelope.ts JSON output | mitigate | `JSON.stringify` correctly escapes; envelope helpers use object construction, never string concatenation. `emitJsonEnvelope` is a single line: `console.log(JSON.stringify(buildEnvelope(args), null, 2))`. |
| T-32-02 | Information Disclosure | help-groups snapshot capturing user-specific paths | mitigate | Snapshot uses `program.helpInformation()` only — Commander does not include cwd, env, or DB paths in help text. Version line stripped before snapshot. |
| T-32-03 | Information Disclosure | Snapshot test capturing `process.env`-derived values | accept | Snapshot covers only help text. Defense-in-depth: if a future contributor adds `.description(\`Memory at ${process.env.X}\`)`, this snapshot would catch it. |
| T-32-04 | Tampering | Emission helper bypass — caller writes bespoke JSON to stdout | mitigate | Plan 02 task instructions REQUIRE all exit paths route through `emitJsonEnvelope` / `emitJsonErrorEnvelope`. Per-command tests assert envelope shape on every branch (success, empty, validation, not-found, catch). |

ASVS L1 V5 (Input Validation) applies: Commander.js `.choices()` (consumed by Plans 32-02 and 32-03) is the validation gate for `--format`. This plan defines no new input surface beyond what Commander already validates.

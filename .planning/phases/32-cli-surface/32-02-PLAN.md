export const ENVELOPE_SCHEMA_VERSION = "1" as const;
export const QUERY_COMMAND_NAMES = ["search", "context", "show", "list", "related", "stats"] as const;
export const QUERY_RESULT_KINDS = ["message", "session", "context", "related", "stats", "file"] as const;
export type QueryCommandName = typeof QUERY_COMMAND_NAMES[number];
export type QueryResultKind = typeof QUERY_RESULT_KINDS[number];
export type EnvelopeScope = { type: "global" } | { type: "project"; project: string };

export function buildEnvelope<T>(args: { command: QueryCommandName; kind: QueryResultKind; data: T; scope?: EnvelopeScope; meta?: Record<string, unknown>; }): QueryResultEnvelope<T>;
export function buildErrorEnvelope(args: { command: QueryCommandName; code: string; message: string; context?: Record<string, unknown>; }): QueryErrorEnvelope;

// SHARED EMISSION HELPERS — every exit path routes through these (Codex HIGH-2):
export function emitJsonEnvelope<T>(args: { command: QueryCommandName; kind: QueryResultKind; data: T; scope?: EnvelopeScope; meta?: Record<string, unknown>; }): void;
export function emitJsonErrorEnvelope(args: { command: QueryCommandName; code: string; message: string; context?: Record<string, unknown>; }): void;

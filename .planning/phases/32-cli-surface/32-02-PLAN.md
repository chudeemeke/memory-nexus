From src/presentation/cli/formatters/envelope.ts:
```typescript
export const ENVELOPE_SCHEMA_VERSION = "1" as const;
export type QueryCommandName = "search" | "context" | "show" | "list" | "related" | "stats";
export type QueryResultKind = "message" | "session" | "context" | "related" | "stats";

export interface QueryResultEnvelope<T = unknown> {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  kind: QueryResultKind;
  scope?: string;
  meta?: Record<string, unknown>;
  data: T;
}

export interface QueryErrorEnvelope {
  schema_version: typeof ENVELOPE_SCHEMA_VERSION;
  command: QueryCommandName;
  error: { code: string; message: string; context?: Record<string, unknown>; };
}

export function buildEnvelope<T>(args: { command: QueryCommandName; kind: QueryResultKind; data: T; scope?: string; meta?: Record<string, unknown>; }): QueryResultEnvelope<T>;
export function buildErrorEnvelope(args: { command: QueryCommandName; code: string; message: string; context?: Record<string, unknown>; }): QueryErrorEnvelope;
```

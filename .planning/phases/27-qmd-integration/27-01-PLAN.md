```typescript
import { spawn } from "node:child_process";
import type { ISummaryGenerator } from "../../domain/ports/index.js";

export class ClaudeSummaryGenerator implements ISummaryGenerator {
  async generateSummary(...): Promise<string> {
    return new Promise((resolve, reject) => {
      const child = spawn("claude", ["-p", "--output-format", "text"], {
        stdio: ["pipe", "pipe", "pipe"],
      });
      let stdout = "";
      let stderr = "";
      child.stdout.on("data", (chunk: Buffer) => { stdout += chunk.toString(); });
      child.stderr.on("data", (chunk: Buffer) => { stderr += chunk.toString(); });
      child.on("error", (err: Error) => { reject(new Error(`Failed to spawn: ${err.message}`)); });
      child.on("close", (code: number | null) => {
        if (code === 0) { resolve(stdout.trim()); }
        else { reject(new Error(`exited with code ${code}: ${stderr.trim()}`)); }
      });
    });
  }
}
```

```typescript
/**
 * AI Formatter Utility
 *
 * Shared utility for producing clean, token-efficient text output.
 * Used by --format ai across all commands and by the smart context
 * service for budget estimation.
 */

// Standard SGR + OSC + cursor movement stripping regex
// Handles: \x1b[...m (SGR), \x1b]...BEL (OSC), \x1b[...letter (CSI)
const ANSI_REGEX = /\x1b\[[0-9;]*[a-zA-Z]|\x1b\].*?\x07|\x1b\(B/g;

export function stripAnsi(text: string): string {
  return text.replace(ANSI_REGEX, "");
}

export function estimateTokens(text: string, charsPerToken = 4): number {
  if (text.length === 0) return 0;
  return Math.ceil(text.length / charsPerToken);
}

export function formatForAi(text: string): string {
  const stripped = stripAnsi(text);
  // Normalize multiple blank lines to single blank line
  const normalized = stripped.replace(/\n{3,}/g, "\n\n");
  return normalized.trim();
}
```

```typescript
/**
 * Memory file types exempt from temporal decay.
 * These files contain evergreen knowledge that remains relevant regardless of age.
 */
export const CURATED_FILE_TYPES: readonly string[] = [
  "decisions",
  "learnings",
  "user_prefs",
];

/**
 * Apply temporal decay with exemptions for curated content.
 *
 * Identical to applyTemporalDecay but skips decay for rowids in the
 * exemptions set. Exempt results retain their original score.
 *
 * @param results Array of results with base scores
 * @param timestamps Map of rowid to message timestamp
 * @param exemptRowids Set of rowids that should not be decayed
 * @param halfLifeDays Number of days for score to halve (default 30)
 * @param now Reference time for age calculation (default: current time)
 * @returns Results with decayedScore, sorted descending
 */
export function applyTemporalDecayWithExemptions<T extends DecayableResult>(
    results: T[],
    timestamps: Map<number, Date>,
    exemptRowids: Set<number>,
    halfLifeDays: number = 30,
    now: Date = new Date()
): Array<T & { decayedScore: number }> {
    const nowMs = now.getTime();
    const msPerDay = 1000 * 60 * 60 * 24;

    const decayed = results.map((r) => {
        // Exempt rowids skip decay entirely
        if (exemptRowids.has(r.rowid)) {
            return { ...r, decayedScore: r.score };
        }

        const timestamp = timestamps.get(r.rowid);
        if (!timestamp) {
            return { ...r, decayedScore: r.score };
        }

        const ageDays = (nowMs - timestamp.getTime()) / msPerDay;
        const decayFactor = Math.pow(0.5, ageDays / halfLifeDays);

        return { ...r, decayedScore: r.score * decayFactor };
    });

    return decayed.sort((a, b) => b.decayedScore - a.decayedScore);
}
```

```typescript
async findCrossProjectLearnings(
    excludeProject?: string,
    limit: number = 20
): Promise<MemoryFile[]> {
    if (excludeProject) {
        const rows = this.db
            .prepare<MemoryFileRow, [string, number]>(
                `SELECT * FROM memory_files
                 WHERE file_type = 'learnings'
                   AND content LIKE '%Applies to: cross-project%'
                   AND (project_encoded IS NULL OR project_encoded != ?)
                 ORDER BY last_indexed_at DESC
                 LIMIT ?`
            )
            .all(excludeProject, limit);
        return rows.map((r) => this.toEntity(r));
    }

    const rows = this.db
        .prepare<MemoryFileRow, [number]>(
            `SELECT * FROM memory_files
             WHERE file_type = 'learnings'
               AND content LIKE '%Applies to: cross-project%'
             ORDER BY last_indexed_at DESC
             LIMIT ?`
        )
        .all(limit);
    return rows.map((r) => this.toEntity(r));
}
```

**Key design decision:** Use SQL `LIKE` on content column directly, NOT FTS5 MATCH. Rationale from research: FTS5 would tokenize "cross-project" into two tokens and return false matches. The result set is small (typically <50 learnings files) so LIKE performance is acceptable.
